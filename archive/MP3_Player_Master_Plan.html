The Missing 5V Boost Converter

You bought a TP4056 and an AMS1117-3.3V, but you are missing a 5V Boost Converter to step up the battery voltage.

A LiPo battery drops from 4.2V down to 3.0V. The ESP32 5V pin needs a solid 5V. If you try to power the ESP32 or the AMS1117-3.3V module directly from the battery, the regulators will drop the voltage by another ~1.1V, resulting in a dead 2.6V system.

Action Required

You must acquire an MT3608 5V Boost Converter module. The battery connects to the TP4056, the TP4056 connects to the MT3608, and the MT3608 outputs a rock-solid 5V to the ESP32 VIN.

SPI TFT & UDA1334A Audio

You purchased a 1.8" Color TFT instead of a basic OLED. This is awesome! We will put it on the same SPI bus as the SD card to save pins.

You also purchased the UDA1334A I2S DAC. This is a brilliant chip. The ESP32 streams pure digital audio to it via I2S, and it decodes it for your wired headphones perfectly.

Perfect Match

We will share the SPI SCK and MOSI pins between the SD card and the TFT, but they will each get their own dedicated CS (Chip Select) pins.

Master Pin Allocation

Carefully selected to avoid all strapping pin conflicts (GPIO 0, 2, 12, 15) and properly share the SPI bus.

The v2.0 Core Firmware

Updated for the 1.8" TFT (Adafruit_ST7735) sharing the SPI bus with the SD card.

Required Arduino Libraries:
1. ESP32-audioI2S (by schreibfaul1)
2. Adafruit ST7735 and ST7789 Library
3. Adafruit GFX Library 
/*
 * DIY Hi-Fi Portable MP3 Player - v2.0 Core System (TFT Edition)
 * Board: ESP32 Dev Module (WROOM-32)
 */

#include <Arduino.h>
#include <WiFi.h>
#include "SPI.h"
#include "SD.h"
#include "Adafruit_GFX.h"
#include "Adafruit_ST7735.h"
#include "Audio.h" // ESP32-audioI2S library

// --- PIN DEFINITIONS ---
// Shared SPI
#define SPI_SCK        18
#define SPI_MOSI       23
#define SPI_MISO       19

// SD Card
#define SD_CS          5

// TFT Screen
#define TFT_CS         21
#define TFT_DC         2
#define TFT_RST        4

// I2S DAC (UDA1334A)
#define I2S_BCLK       26
#define I2S_LRC        25
#define I2S_DOUT       22

// Buttons (Active LOW)
#define BTN_PLAYPAUSE  32
#define BTN_NEXT       33
#define BTN_PREV       27

// --- GLOBALS ---
Audio audio;
Adafruit_ST7735 tft = Adafruit_ST7735(TFT_CS, TFT_DC, TFT_MOSI, SPI_SCK, TFT_RST);
int volume = 15; // 0 to 21
bool isPlaying = false;
unsigned long lastButtonPress = 0;

void setup() {
    Serial.begin(115200);
    WiFi.mode(WIFI_OFF); // Save battery, prevent static

    // 1. Initialize Buttons
    pinMode(BTN_PLAYPAUSE, INPUT_PULLUP);
    pinMode(BTN_NEXT, INPUT_PULLUP);
    pinMode(BTN_PREV, INPUT_PULLUP);

    // 2. Initialize SPI & TFT
    SPI.begin(SPI_SCK, SPI_MISO, SPI_MOSI, TFT_CS); // Initialize SPI bus
    tft.initR(INITR_BLACKTAB); // Init ST7735S chip
    tft.setRotation(1); // Landscape
    tft.fillScreen(ST77XX_BLACK);
    
    tft.setTextColor(ST77XX_WHITE);
    tft.setTextSize(1);
    tft.setCursor(10, 20);
    tft.println("Booting System...");

    // 3. Initialize SD Card
    // Important: Re-assign CS pin for SD card on the same SPI bus
    if(!SD.begin(SD_CS, SPI, 4000000)) {
        Serial.println("SD Card Mount Failed!");
        tft.setTextColor(ST77XX_RED);
        tft.setCursor(10, 40);
        tft.println("SD Error! Check wiring.");
        while(1);
    }

    // 4. Initialize Audio
    audio.setPinout(I2S_BCLK, I2S_LRC, I2S_DOUT);
    audio.setVolume(volume);

    // Play first track (Example)
    audio.connecttoFS(SD, "/track1.mp3");
    isPlaying = true;

    updateDisplay("track1.mp3");
}

void loop() {
    // 1. Audio Processing (Must be called constantly)
    audio.loop();

    // 2. Handle Buttons
    if (millis() - lastButtonPress > 250) {
        if (digitalRead(BTN_PLAYPAUSE) == LOW) {
            audio.pauseResume();
            isPlaying = !isPlaying;
            updateDisplay("track1.mp3");
            lastButtonPress = millis();
        }
    }
}

// --- HELPER FUNCTIONS ---
void updateDisplay(String trackName) {
    tft.fillScreen(ST77XX_BLACK);
    
    // Status Bar
    tft.fillRect(0, 0, 160, 15, ST77XX_BLUE);
    tft.setTextColor(ST77XX_WHITE);
    tft.setCursor(5, 4);
    tft.print("Vol: "); tft.print(volume);
    
    // Track Info
    tft.setTextColor(ST77XX_CYAN);
    tft.setTextSize(1);
    tft.setCursor(5, 30);
    tft.println("Now Playing:");
    
    tft.setTextColor(ST77XX_WHITE);
    tft.setTextSize(2);
    tft.setCursor(5, 45);
    tft.println(trackName.substring(0, 10)); 
    
    // Play/Pause icon
    tft.setTextSize(1);
    tft.setCursor(5, 90);
    if(isPlaying) {
        tft.setTextColor(ST77XX_GREEN);
        tft.print(" [ > PLAYING ]");
    } else {
        tft.setTextColor(ST77XX_YELLOW);
        tft.print(" [ || PAUSED ]");
    }
}

// Callbacks
void audio_info(const char *info){ Serial.println(info); }
